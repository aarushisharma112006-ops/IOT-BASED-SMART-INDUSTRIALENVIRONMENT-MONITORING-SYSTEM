// ------------ SEND WEBPAGE ------------
void handleRoot() {
  server.send_P(200, "text/html", webpage);
}

// ------------ SEND SENSOR DATA AS JSON ------------
void handleData() {
  String json = "{";

  json += "\"temperature_c\":";
  if (isnan(temperatureC)) json += "null";
  else json += String(temperatureC, 1);

  json += ",\"humidity_percent\":";
  if (isnan(humidity)) json += "null";
  else json += String(humidity, 1);

  json += ",\"ldr_raw\":" + String(lightValue);
  json += ",\"mq_raw\":" + String(mqValue);
  json += ",\"rain_raw\":" + String(rainValue);

  json += ",\"intense_light\":";
  json += (intenseLight ? "true" : "false");

  json += ",\"gas_alert\":";
  json += (gasAlert ? "true" : "false");

  json += ",\"rain_detected\":";
  json += (rainDetected ? "true" : "false");

  json += "}";

  server.send(200, "application/json", json);
}

// ------------ READ SENSORS ------------
void readSensors() {
  temperatureC = dht.readTemperature();
  humidity = dht.readHumidity();

  rainValue = analogRead(RAIN_PIN);
  lightValue = analogRead(LDR_PIN);
  mqValue = analogRead(MQ_AO);

  // These conditions assume your sensor readings follow
  // the directions observed in your previous tests.
  rainDetected = (rainValue < rainThreshold);
  intenseLight = (lightValue > ldrThreshold);
  gasAlert = (mqValue > mqThreshold);

  // Only the LDR condition controls the red LED and buzzer.
  if (intenseLight) {
    digitalWrite(RED_LED, HIGH);
    digitalWrite(GREEN_LED, LOW);
    digitalWrite(BUZZER, HIGH);
  } else {
    digitalWrite(RED_LED, LOW);
    digitalWrite(GREEN_LED, HIGH);
    digitalWrite(BUZZER, LOW);
  }

  Serial.println("----- SENSOR READINGS -----");
  Serial.print("Temperature: ");
  if (isnan(temperatureC)) Serial.println("DHT read error");
  else Serial.print(temperatureC), Serial.println(" C");

  Serial.print("Humidity: ");
  if (isnan(humidity)) Serial.println("DHT read error");
  else Serial.print(humidity), Serial.println(" %");

  Serial.print("LDR raw: ");
  Serial.println(lightValue);

  Serial.print("MQ raw: ");
  Serial.println(mqValue);

  Serial.print("Rain raw: ");
  Serial.println(rainValue);

  Serial.print("Intense light: ");
  Serial.println(intenseLight ? "YES" : "NO");
}

// ------------ UPDATE OLED ------------
void updateOLED() {
  if (!oledFound) return;

  display.clearDisplay();
  display.setTextSize(1);
  display.setTextColor(SSD1306_WHITE);
  display.setCursor(0, 0);

  display.println("Industrial Monitor");

  display.print("Temp: ");
  if (isnan(temperatureC)) display.println("--");
  else {
    display.print(temperatureC, 1);
    display.println(" C");
  }

  display.print("Humidity: ");
  if (isnan(humidity)) display.println("--");
  else {
    display.print(humidity, 1);
    display.println(" %");
  }

  display.print("LDR: ");
  display.println(lightValue);

  display.print("MQ: ");
  display.println(mqValue);

  display.print("Rain: ");
  display.println(rainValue);

  display.print("Light: ");
  display.println(intenseLight ? "INTENSE!" : "Normal");

  display.display();
}

// ------------ SETUP ------------
void setup() {
  Serial.begin(115200);

  pinMode(GREEN_LED, OUTPUT);
  pinMode(RED_LED, OUTPUT);
  pinMode(BUZZER, OUTPUT);

  digitalWrite(GREEN_LED, LOW);
  digitalWrite(RED_LED, LOW);
  digitalWrite(BUZZER, LOW);

  analogReadResolution(12);

  dht.begin();

  Wire.begin(OLED_SDA, OLED_SCL);
  oledFound = display.begin(SSD1306_SWITCHCAPVCC, 0x3C);

  if (!oledFound) {
    Serial.println("OLED not detected at address 0x3C.");
    Serial.println("Webpage and sensors can still work.");
  }

  WiFi.mode(WIFI_STA);
  WiFi.begin(ssid, password);

  Serial.print("Connecting to Wi-Fi");

  unsigned long wifiStart = millis();
  while (WiFi.status() != WL_CONNECTED &&
         millis() - wifiStart < 20000) {
    delay(500);
    Serial.print(".");
  }

  Serial.println();

  if (WiFi.status() == WL_CONNECTED) {
    Serial.println("Wi-Fi connected!");
    Serial.print("Open this webpage: http://");
    Serial.println(WiFi.localIP());
  } else {
    Serial.println("Wi-Fi connection failed.");
    Serial.println("Check the Wi-Fi name and password, then reset.");
  }

  server.on("/", HTTP_GET, handleRoot);
  server.on("/data", HTTP_GET, handleData);
  server.begin();

  Serial.println("Web server started.");

  readSensors();
  updateOLED();
  lastSensorRead = millis();
}

// ------------ MAIN LOOP ------------
void loop() {
  server.handleClient();

  if (millis() - lastSensorRead >= sensorInterval) {
    lastSensorRead = millis();

    readSensors();
    updateOLED();
  }

  delay(2);
}
